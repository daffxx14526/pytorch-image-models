{==================================================================}
{ 机构资金雷达 V3 - 强制四色版(无灰色)                                      }
{ 若仍见灰色或震荡二字, 请删除旧公式后全新粘贴本文件                        }
{ 绿=出货 黄=洗盘 红=吸筹 紫=拉升                                          }
{==================================================================}

N1:=5;
N2:=20;
N3:=20;
N4:=5;
涨幅阈值:=3;
量增阈值:=1.2;
关注线:=70;
加速线:=85;
启用L2:=1;
L2权重:=0.12;
L2模式:=1;

涨幅:=(C-REF(C,1))/REF(C,1)*100;
量比:=VOL/MA(VOL,N1);
均量:=MA(VOL,N2);
换手率:=IF(FINANCE(7)>0,VOL/FINANCE(7)*100,0);
均换手:=MA(换手率,N2);
振幅:=(H-L)/REF(C,1)*100;
均价:=IF(VOL>0,AMOUNT/(VOL*100),C);
MA20:=MA(C,20);
MA60:=MA(C,60);

OBV增量:=IF(C>REF(C,1),VOL,IF(C<REF(C,1),-VOL,0));
OBV值:=SUM(OBV增量,0);
OBV新高:=OBV值>=HHV(OBV值,N3);

估价中枢:=(H+L+C)/3;
上影幅:=IF(H=L,0,H-MAX(C,O));
下影幅:=IF(H=L,0,MIN(C,O)-L);
实体上幅:=IF(H=L,0,MAX(C,O)-估价中枢);
实体下幅:=IF(H=L,0,估价中枢-MIN(C,O));
单位量:=VOL/IF(H=L,4,H-L+0.0001);
估外盘:=(实体下幅+下影幅)*单位量;
估内盘:=(上影幅+实体上幅)*单位量;

主动买:=ACTINVOL;
主动卖:=ACTOUTVOL;
大单买:=LARGEINTRDVOL;
大单卖:=LARGEOUTTRADEVOL;
有主动:=主动买+主动卖>0;
有大单:=大单买+大单卖>0;
外盘:=IF(L2模式=2 AND 有大单,大单买,IF(有主动,主动买,估外盘));
内盘:=IF(L2模式=2 AND 有大单,大单卖,IF(有主动,主动卖,估内盘));
L2总量:=外盘+内盘;
L2净比:=IF(L2总量>0,(外盘-内盘)/L2总量,0);
L2买卖比:=IF(内盘>0,外盘/内盘,1);
L2净均:=MA(L2净比,N4);
L2大买:=L2净比>0.05;
L2大卖:=L2净比<-0.05;
L2持续买:=L2净比>0 AND REF(L2净比,1)>0 AND REF(L2净比,2)>0;
L2持续卖:=L2净比<0 AND REF(L2净比,1)<0 AND REF(L2净比,2)<0;
L2加速买:=L2净比>REF(L2净比,1) AND REF(L2净比,1)>REF(L2净比,2) AND L2净比>0;
L2加速卖:=L2净比<REF(L2净比,1) AND REF(L2净比,1)<REF(L2净比,2) AND L2净比<0;
有L2数据:=有主动 OR 有大单;

S1:=IF(量比>=量增阈值 AND 涨幅<涨幅阈值 AND 涨幅>-2,1,IF(量比>=1.1 AND 涨幅<5,0.5,IF(量比>1.1 AND 涨幅<0,-0.5,0)));
S2:=IF(量比<=0.7 AND 涨幅>-1,0.8,IF(量比<=0.8 AND 涨幅>=0,0.5,IF(量比<=0.6 AND 涨幅<-3,-0.5,0)));
S3:=IF(涨幅>5 AND 换手率<均换手*0.8,1,IF(涨幅>3 AND 换手率<均换手,0.5,IF(涨幅>5 AND 换手率>均换手*2,-0.8,0)));
S4:=IF(OBV新高,1,IF(OBV值>REF(OBV值,1),0.3,IF(OBV值<=LLV(OBV值,N3),-0.5,0)));
S5:=IF(换手率>均换手*1.5 AND 振幅<5,0.6,IF(换手率>均换手*2 AND 振幅<3,0.8,0));
S6:=IF(C>均价,0.8,IF(C<均价*0.98,-0.5,0));
小阳:=C>REF(C,1) AND 涨幅>0 AND 涨幅<3;
量递增:=VOL>REF(VOL,1) AND REF(VOL,1)>REF(VOL,2);
S7:=IF(小阳 AND REF(小阳,1) AND REF(小阳,2) AND 量递增,1,0);

SL2A:=IF(L2买卖比>1.2,0.8,IF(L2买卖比>1.05,0.4,IF(L2买卖比<0.8,-0.8,IF(L2买卖比<0.95,-0.4,0))));
SL2B:=IF(L2净比>0.05,0.8,IF(L2净比>0.02,0.4,IF(L2净比<-0.05,-0.8,IF(L2净比<-0.02,-0.4,0))));
SL2C:=IF(L2加速买,0.7,IF(L2加速卖,-0.7,IF(L2净比>L2净均 AND L2净比>0,0.3,IF(L2净比<L2净均 AND L2净比<0,-0.3,0))));
SL2:=SL2A*0.45+SL2B*0.40+SL2C*0.15;

跌势罚:=IF(涨幅<-3,-0.15,IF(涨幅<-1,-0.08,IF(涨幅>3,0.05,0)));
L2修正:=IF(启用L2=1 AND L2大卖 AND 涨幅<4 AND 量比>1.0,-0.12,IF(启用L2=1 AND L2大买 AND L2持续买,0.06,IF(启用L2=1 AND L2持续卖,-0.08,0)));
原始分:=IF(启用L2=1,S1*0.22+S2*0.18+S3*0.13+S4*0.13+S5*0.09+S6*0.09+S7*0.04+SL2*L2权重+L2修正+跌势罚,S1*0.25+S2*0.20+S3*0.15+S4*0.15+S5*0.10+S6*0.10+S7*0.05+跌势罚);

资金强度:(原始分+1)*50,NODRAW;
强度均线:MA(资金强度,3),COLORWHITE,LINETHICK2;
L2强度线:IF(启用L2=1,L2净比*50+50,50),COLOR00FFFF,LINETHICK1;

关注:关注线,COLOR808080,DOTLINE;
加速:加速线,COLOR808080,DOTLINE;
中轴:50,COLOR404040,DOTLINE;

放量下跌:=量比>1.2 AND 涨幅<-1.5;
高位大阴:=C<REF(C,1) AND (REF(C,1)-C)/REF(C,1)*100>2.5 AND C>(MA60*1.03);
L2强卖:=启用L2=1 AND (L2大卖 OR L2持续卖 OR L2净比<-0.02);
L2强买:=启用L2=1 AND (L2大买 OR L2持续买 OR L2净比>0.02);
出货日:=放量下跌 OR 高位大阴 OR (涨幅<-0.8 AND 量比>1.05) OR L2强卖;

{--- 强制四色: 每根K线必为绿/黄/红/紫之一 ---}
是绿:=出货日 OR 资金强度<50;
是紫:=资金强度>=80 AND 是绿=0;
是红:=资金强度>=65 AND 资金强度<80 AND 是绿=0;
是黄:=是绿=0 AND 是紫=0 AND 是红=0;

STICKLINE(是绿,资金强度,0,4,0),COLOR00CC00;
STICKLINE(是紫,资金强度,0,4,0),COLORMAGENTA;
STICKLINE(是红,资金强度,0,4,0),COLORRED;
STICKLINE(是黄,资金强度,0,4,0),COLORYELLOW;

PHASE:=IF(是绿,1,IF(是紫,4,IF(是红,3,2)));

上穿关注:=CROSS(资金强度,关注线);
红转紫:=PHASE=4 AND REF(PHASE,1)=3;
紫转绿:=PHASE=1 AND REF(PHASE,1)=4;
L2共振买:=L2强买 AND PHASE=3;
L2共振卖:=L2强卖 AND PHASE=1;

DRAWICON(上穿关注,资金强度,1);
DRAWTEXT(上穿关注,资金强度*0.95,'关注'),COLORRED;
DRAWTEXT(红转紫,资金强度*1.02,'加速'),COLORMAGENTA;
DRAWTEXT(紫转绿,资金强度*0.90,'减仓'),COLOR00CC00;
DRAWTEXT(L2共振买,资金强度*1.08,'L2买'),COLORRED;
DRAWTEXT(L2共振卖,资金强度*0.85,'L2卖'),COLOR00CC00;

DRAWTEXT(ISLASTBAR AND PHASE=1,资金强度,'出货'),COLOR00CC00;
DRAWTEXT(ISLASTBAR AND PHASE=2,资金强度,'洗盘'),COLORYELLOW;
DRAWTEXT(ISLASTBAR AND PHASE=3,资金强度,'吸筹'),COLORRED;
DRAWTEXT(ISLASTBAR AND PHASE=4,资金强度,'拉升'),COLORMAGENTA;
DRAWTEXT(ISLASTBAR AND 有L2数据,资金强度*0.88,'L2'),COLOR00FFFF;
DRAWTEXT(ISLASTBAR AND 有L2数据=0,资金强度*0.88,'估价'),COLORYELLOW;
DRAWTEXT(ISLASTBAR,资金强度*0.78,'V3'),COLORWHITE;
