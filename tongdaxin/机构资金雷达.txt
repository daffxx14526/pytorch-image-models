{==================================================================}
{ 机构资金雷达 - 主副图指标                                              }
{ 基于公开行情多维度推断主力资金行为，不使用未来函数                      }
{ 四色阶段: 绿=出货 黄=洗盘 红=吸筹 紫=拉升                            }
{ 参数说明见 tongdaxin/README.md                                       }
{==================================================================}

{--- 可调参数 ---}
N1:=5;
N2:=20;
N3:=20;
涨幅阈值:=3;
量增阈值:=1.2;
关注线:=70;
加速线:=85;

{--- 基础变量 ---}
涨幅:=(C-REF(C,1))/REF(C,1)*100;
量比:=VOL/MA(VOL,N1);
均量:=MA(VOL,N2);
换手率:=IF(FINANCE(7)>0,VOL/FINANCE(7)*100,0);
均换手:=MA(换手率,N2);
振幅:=(H-L)/REF(C,1)*100;
均振幅:=MA(振幅,N2);
均价:=IF(VOL>0,AMOUNT/(VOL*100),C);
MA20:=MA(C,20);
MA60:=MA(C,60);

{--- OBV 能量潮 ---}
OBV增量:=IF(C>REF(C,1),VOL,IF(C<REF(C,1),-VOL,0));
OBV值:=SUM(OBV增量,0);
OBV新高:=OBV值>=HHV(OBV值,N3);

{--- 七维度子评分 (-1 ~ +1) ---}

{1. 量价背离 25%: 量增价平/小涨}
S1:=IF(量比>=量增阈值 AND 涨幅<涨幅阈值 AND 涨幅>-2,1,
    IF(量比>=1.1 AND 涨幅<5,0.5,
    IF(量比>1.1 AND 涨幅<0,-0.5,0)));

{2. 锁仓度 20%: 缩量不跌}
S2:=IF(量比<=0.7 AND 涨幅>-1,0.8,
    IF(量比<=0.8 AND 涨幅>=0,0.5,
    IF(量比<=0.6 AND 涨幅<-3,-0.5,0)));

{3. 控盘度 15%: 大阳低换手}
S3:=IF(涨幅>5 AND 换手率<均换手*0.8,1,
    IF(涨幅>3 AND 换手率<均换手,0.5,
    IF(涨幅>5 AND 换手率>均换手*2,-0.8,0)));

{4. OBV 趋势 15%}
S4:=IF(OBV新高,1,
    IF(OBV值>REF(OBV值,1),0.3,
    IF(OBV值<=LLV(OBV值,N3),-0.5,0)));

{5. 振幅-换手 10%: 高换低振=筹码交换}
S5:=IF(换手率>均换手*1.5 AND 振幅<5,0.6,
    IF(换手率>均换手*2 AND 振幅<3,0.8,0));

{6. 尾盘强度 10%: 收盘相对均价}
S6:=IF(C>均价,0.8,
    IF(C<均价*0.98,-0.5,0));

{7. 连续形态 5%: 三日小阳温和放量}
小阳:=C>REF(C,1) AND 涨幅>0 AND 涨幅<3;
量递增:=VOL>REF(VOL,1) AND REF(VOL,1)>REF(VOL,2);
S7:=IF(小阳 AND REF(小阳,1) AND REF(小阳,2) AND 量递增,1,0);

{--- 加权综合评分 0~100 ---}
原始分:=S1*0.25+S2*0.20+S3*0.15+S4*0.15+S5*0.10+S6*0.10+S7*0.05;
资金强度:(原始分+1)*50,NODRAW;
强度均线:MA(资金强度,3),COLORWHITE,LINETHICK1;

{--- 参考线 ---}
关注:关注线,COLOR808080,DOTLINE;
加速:加速线,COLOR808080,DOTLINE;
中轴:50,COLOR404040,DOTLINE;

{--- 四色阶段判定 (互斥优先级) ---}
{PHASE: 1=出货绿 2=洗盘黄 3=吸筹红 4=拉升紫 0=震荡灰}

放量下跌:=量比>1.3 AND 涨幅<-2;
高位大阴:=C<REF(C,1) AND (REF(C,1)-C)/REF(C,1)*100>3 AND C>MA60*1.05;
量波动:=ABS(VOL-均量)/均量>0.3;
振幅收窄:=振幅<均振幅*0.8;

PHASE:=IF(资金强度>=关注线 AND OBV新高 AND C>MA20,4,
    IF(资金强度>=55 AND 振幅收窄 AND 量波动,2,
    IF(资金强度<=35 AND (放量下跌 OR 高位大阴),1,
    IF(资金强度>=45 AND 量比>=量增阈值 AND 涨幅<涨幅阈值,3,0))));

{--- 四色柱状图 ---}
STICKLINE(PHASE=1,资金强度,0,3,0),COLOR00AA00;
STICKLINE(PHASE=2,资金强度,0,3,0),COLORYELLOW;
STICKLINE(PHASE=3,资金强度,0,3,0),COLORRED;
STICKLINE(PHASE=4,资金强度,0,3,0),COLORMAGENTA;
STICKLINE(PHASE=0,资金强度,0,3,0),COLORGRAY;

{--- 异动标记 ---}
上穿关注:=CROSS(资金强度,关注线);
红转紫:=PHASE=4 AND REF(PHASE,1)=3;
紫转绿:=PHASE=1 AND REF(PHASE,1)=4 AND 放量下跌;

DRAWICON(上穿关注,资金强度,1);
DRAWTEXT(上穿关注,资金强度*0.95,'↑关注'),COLORRED;
DRAWTEXT(红转紫,资金强度*1.02,'加速'),COLORMAGENTA;
DRAWTEXT(紫转绿,资金强度*0.90,'减仓'),COLOR00AA00;

{--- 阶段文字 (副图左上角提示) ---}
DRAWTEXT(ISLASTBAR AND PHASE=1,资金强度,'出货'),COLOR00AA00;
DRAWTEXT(ISLASTBAR AND PHASE=2,资金强度,'洗盘'),COLORYELLOW;
DRAWTEXT(ISLASTBAR AND PHASE=3,资金强度,'吸筹'),COLORRED;
DRAWTEXT(ISLASTBAR AND PHASE=4,资金强度,'拉升'),COLORMAGENTA;
DRAWTEXT(ISLASTBAR AND PHASE=0,资金强度,'震荡'),COLORGRAY;
